\subsection{Advanced Mathematical Constructs}

\begin{definition}[Riemann Integral]
A function $f:[a,b] \to \mathbb{R}$ is Riemann integrable if
\begin{equation}
\lim_{\|\Delta\| \to 0} \sum_{i=1}^n f(\xi_i)\Delta x_i = \int_a^b f(x)\,dx
\end{equation}
exists, where $\Delta = \{x_0, x_1, \ldots, x_n\}$ is a partition of $[a,b]$ and $\xi_i \in [x_{i-1}, x_i]$.
\end{definition}

\begin{theorem}[Fundamental Theorem of Calculus]
If $f$ is continuous on $[a,b]$ and $F(x) = \int_a^x f(t)\,dt$, then:
\begin{enumerate}
\item $F$ is differentiable on $(a,b)$ and $F'(x) = f(x)$
\item $\int_a^b f(x)\,dx = G(b) - G(a)$ for any antiderivative $G$ of $f$
\end{enumerate}
\end{theorem}

\begin{proof}
For part (1), consider the difference quotient:
\begin{align}
\frac{F(x+h) - F(x)}{h} &= \frac{1}{h}\left(\int_a^{x+h} f(t)\,dt - \int_a^x f(t)\,dt\right) \\
&= \frac{1}{h}\int_x^{x+h} f(t)\,dt
\end{align}

By the Mean Value Theorem for integrals, there exists $c \in [x, x+h]$ such that:
$$\int_x^{x+h} f(t)\,dt = f(c) \cdot h$$

Therefore:
$$\lim_{h \to 0} \frac{F(x+h) - F(x)}{h} = \lim_{h \to 0} f(c) = f(x)$$
since $f$ is continuous and $c \to x$ as $h \to 0$.
\end{proof}

\subsection{Formal Algorithm Descriptions}

\begin{algorithm}
\caption{Advanced Quicksort with Median-of-Three Pivot Selection}
\label{alg:quicksort}
\begin{algorithmic}[1]
\REQUIRE Array $A[1..n]$ of comparable elements
\ENSURE Array $A$ sorted in non-decreasing order

\PROCEDURE{QuickSort}{$A, low, high$}
    \IF{$low < high$}
        \STATE $pivot \leftarrow$ \CALL{MedianOfThree}{$A, low, high$}
        \STATE $p \leftarrow$ \CALL{Partition}{$A, low, high, pivot$}
        \STATE \CALL{QuickSort}{$A, low, p-1$}
        \STATE \CALL{QuickSort}{$A, p+1, high$}
    \ENDIF
\ENDPROCEDURE

\PROCEDURE{MedianOfThree}{$A, low, high$}
    \STATE $mid \leftarrow \lfloor(low + high)/2\rfloor$
    \IF{$A[mid] < A[low]$}
        \STATE \CALL{Swap}{$A[low], A[mid]$}
    \ENDIF
    \IF{$A[high] < A[low]$}
        \STATE \CALL{Swap}{$A[low], A[high]$}
    \ENDIF
    \IF{$A[high] < A[mid]$}
        \STATE \CALL{Swap}{$A[mid], A[high]$}
    \ENDIF
    \RETURN $A[mid]$
\ENDPROCEDURE

\PROCEDURE{Partition}{$A, low, high, pivot$}
    \STATE $i \leftarrow low - 1$
    \FOR{$j \leftarrow low$ \TO $high - 1$}
        \IF{$A[j] \leq pivot$}
            \STATE $i \leftarrow i + 1$
            \STATE \CALL{Swap}{$A[i], A[j]$}
        \ENDIF
    \ENDFOR
    \STATE \CALL{Swap}{$A[i+1], A[high]$}
    \RETURN $i + 1$
\ENDPROCEDURE
\end{algorithmic}
\end{algorithm}

class Point3D:

    y: float
    z: float
    
    def distance_to(self, other: 'Point3D') -> float:
        """Calculate Euclidean distance to another point."""
        return np.sqrt((self.x - other.x)**2 +

                      (self.z - other.z)**2)
    
    def __add__(self, other: 'Point3D') -> 'Point3D':
        return Point3D(self.x + other.x, 
                      self.y + other.y, 
                      self.z + other.z)

class KDTree:
    """K-dimensional tree for efficient nearest neighbor search."""
    
    def __init__(self, points: List[Point3D], depth: int = 0):
        if not points:
            self.node = None
            return
            
        k = 3  # 3D space
        axis = depth % k
        
        # Sort points by current axis
        if axis == 0:
            points.sort(key=lambda p: p.x)
        elif axis == 1:
            points.sort(key=lambda p: p.y)
        else:
            points.sort(key=lambda p: p.z)
        
        median = len(points) // 2
        self.node = points[median]
        self.left = KDTree(points[:median], depth + 1)
        self.right = KDTree(points[median + 1:], depth + 1)

#include <vector>
#include <algorithm>
#include <memory>
#include <cmath>

template<typename T, size_t Dimensions>
class Point {
private:
    std::array<T, Dimensions> coords;
    
public:
    Point() = default;
    
    template<typename... Args>
    Point(Args... args) : coords{static_cast<T>(args)...} {
        static_assert(sizeof...(args) == Dimensions, 
                     "Number of arguments must match dimensions");
    }
    
    T& operator[](size_t index) { return coords[index]; }
    const T& operator[](size_t index) const { return coords[index]; }
    
    T distance_squared(const Point& other) const {
        T sum = T{0};
        for (size_t i = 0; i < Dimensions; ++i) {
            T diff = coords[i] - other.coords[i];
            sum += diff * diff;
        }
        return sum;
    }
    
    T distance(const Point& other) const {
        return std::sqrt(distance_squared(other));
    }
};

template<typename T, size_t Dimensions>
class KDTree {
private:
    struct Node {
        Point<T, Dimensions> point;
        std::unique_ptr<Node> left, right;
        size_t axis;
        
        Node(const Point<T, Dimensions>& p, size_t a) 
            : point(p), axis(a) {}
    };
    
    std::unique_ptr<Node> root;
    
    std::unique_ptr<Node> build_tree(
        std::vector<Point<T, Dimensions>>& points,
        size_t depth = 0) {
        
        if (points.empty()) return nullptr;
        
        size_t axis = depth % Dimensions;
        
        std::sort(points.begin(), points.end(),
                 [axis](const auto& a, const auto& b) {
                     return a[axis] < b[axis];
                 });
        
        size_t median = points.size() / 2;
        auto node = std::make_unique<Node>(points[median], axis);
        
        std::vector<Point<T, Dimensions>> left_points(
            points.begin(), points.begin() + median);
        std::vector<Point<T, Dimensions>> right_points(
            points.begin() + median + 1, points.end());
        
        node->left = build_tree(left_points, depth + 1);
        node->right = build_tree(right_points, depth + 1);
        
        return node;
    }
    
public:
    void insert(std::vector<Point<T, Dimensions>> points) {
        root = build_tree(points);
    }
};
\end{lstlisting}
\end{codebox}
